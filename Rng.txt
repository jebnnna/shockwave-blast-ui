-- NEW: timestamp helper (works without os.date on restricted executors)
local function timestamp()
    if os and os.date then
        return os.date("%H:%M:%S")
    end
    local t = math.floor(tick() % 86400)
    return string.format("%02d:%02d:%02d", math.floor(t/3600), math.floor((t%3600)/60), t%60)
end

local function checkRare(str)
    local low = str:lower()
    local bestW, bestS = nil, 0
    for word, score in pairs(CFG.RARE_WORDS) do
        if low:find(word) and score > bestS then
            bestW, bestS = word, score
        end
    end
    return bestW, bestS
end

-- NEW: processResult now timestamps and records every result to S.rollHistory
local function processResult(...)
    local parts = {...}
    local combined = ""
    for _, v in ipairs(parts) do combined = combined .. tostring(v) .. " " end
    combined = combined:gsub("^%s+",""):gsub("%s+$","")
    if combined == "" then return end

    local ts   = timestamp()
    local word, score = checkRare(combined)
    local marker = word and "★ " or "› "

    -- record to history (unbounded — export via Copy Log)
    table.insert(S.rollHistory, string.format("[%s] #%d %s%s",
        ts, S.rollCount, marker, combined:sub(1, 80)))

    if word then
        S.hitCount = S.hitCount + 1
        updateCount()
        local display = combined:sub(1, 50)
        log("[" .. ts .. "] ★ " .. display, 255, 210, 0)
        if score > S.bestScore then
            S.bestScore = score
            S.bestCard  = display
            BLbl.Text   = "🏆  Best: " .. display
        end
    else
        log("[" .. ts .. "] › " .. combined:sub(1, 50), 155, 155, 175)
    end
end

-- ─── DRAG — FIX ───────────────────────────────────────────────────────────────
-- Original used TBar.InputChanged which drops events when cursor leaves the bar.
-- Fix: bind Changed and Ended to UserInputService globally.
do
    local dragging, dStart, wStart
    TBar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dStart   = i.Position
            wStart   = Win.Position
        end
    end)
    UIS.InputChanged:Connect(function(i)       -- ← global: fires even off TBar
        if dragging and (i.UserInputType == Enum.UserInputType.Touch
        or  i.UserInputType == Enum.UserInputType.MouseMove) then
            local d = i.Position - dStart
            Win.Position = UDim2.new(
                wStart.X.Scale, wStart.X.Offset + d.X,
                wStart.Y.Scale, wStart.Y.Offset + d.Y)
        end
    end)
    UIS.InputEnded:Connect(function(i)         -- ← global: catches release off TBar
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
end

-- ─── TEST GAME ────────────────────────────────────────────────────────────────
BtnTest.MouseButton1Click:Connect(function()
    setStatus("Scanning game…", 255, 200, 0)
    log("── Scanning for remotes ──", 100, 140, 255)
    task.wait(0.05)

    local matched, allFound = {}, {}
    local function scan(parent)
        for _, obj in ipairs(parent:GetDescendants()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                table.insert(allFound, obj)
                local n = obj.Name:lower()
                for _, kw in ipairs(CFG.ROLL_KEYS) do
                    if n:find(kw) then table.insert(matched, obj); break end
                end
            end
        end
    end
    scan(RS); scan(game.Workspace)
    log("Total remotes in game: " .. #allFound, 170, 170, 170)

    if #matched > 0 then
        S.remote      = matched[1]
        S.remoteClass = matched[1].ClassName
        setStatus("✓ Compatible! Remote: " .. matched[1].Name, 55, 215, 75)
        log("✓ Roll remote: " .. matched[1].Name .. " [" .. matched[1].ClassName .. "]", 60, 220, 100)
        for i = 2, math.min(#matched, 5) do
            log("  alt: " .. matched[i].Name, 130, 200, 130)
        end
        if matched[1]:IsA("RemoteEvent") then
            matched[1].OnClientEvent:Connect(function(...) processResult(...) end)
            log("  Result listener attached.", 120, 120, 160)
        end
    elseif #allFound > 0 then
        setStatus("⚠  Remotes exist, no match", 255, 155, 0)
        log("⚠ No roll keyword matched. All remotes:", 255, 160, 60)
        for i, r in ipairs(allFound) do
            if i > 15 then log("  … and " .. (#allFound-15) .. " more", 140,140,140); break end
            log("  [" .. i .. "] " .. r.Name .. " (" .. r.ClassName .. ")", 200, 200, 200)
        end
        log("→ Falling back to remote [1]: " .. allFound[1].Name, 200, 180, 100)
        S.remote      = allFound[1]
        S.remoteClass = allFound[1].ClassName
        if allFound[1]:IsA("RemoteEvent") then
            allFound[1].OnClientEvent:Connect(function(...) processResult(...) end)
        end
    else
        setStatus("✗  No remotes found", 215, 55, 55)
        log("✗ No RemoteEvents in this game.", 215, 75, 75)
        log("  Game may use local-only RNG — Hook RNG still works.", 155, 155, 155)
    end
end)

-- ─── AUTO ROLL ────────────────────────────────────────────────────────────────
BtnRoll.MouseButton1Click:Connect(function()
    S.autoRoll = not S.autoRoll
    if S.autoRoll then
        BtnRoll.BackgroundColor3 = Color3.fromRGB(155, 50, 35)
        BtnRoll.Text             = "⏸  Stop Roll"
        log("── Auto-roll ON ──", 90, 215, 90)
        if not S.remote then
            log("⚠ No remote detected — run Test Game first.", 255, 160, 50)
            log("  Will keep retrying each tick.", 160, 160, 160)
        end
        task.spawn(function()
            while S.autoRoll do
                if S.remote then
                    pcall(function()
                        if S.remoteClass == "RemoteEvent" then
                            S.remote:FireServer()
                        else
                            S.remote:InvokeServer()
                        end
                    end)
                    S.rollCount = S.rollCount + 1
                    updateCount()
                else
                    local r = RS:FindFirstChildWhichIsA("RemoteEvent", true)
                    if r then S.remote = r; S.remoteClass = r.ClassName end
                end
                task.wait(CFG.ROLL_DELAY)
            end
        end)
    else
        BtnRoll.BackgroundColor3 = Color3.fromRGB(35, 115, 55)
        BtnRoll.Text             = "▶  Auto Roll"
        log("── Auto-roll OFF — total rolls: " .. S.rollCount .. " ──", 190, 190, 190)
    end
end)

-- ─── COPY LOG — NEW ───────────────────────────────────────────────────────────
-- Dumps full unbounded session history to clipboard via setclipboard().
-- Falls back to showing last 10 in the log panel if setclipboard isn't available.
BtnCopy.MouseButton1Click:Connect(function()
    if #S.rollHistory == 0 then
        log("No rolls recorded yet.", 160, 160, 160)
        return
    end
    local header = {
        "VANTA RNG — Session Export",
        "Rolls: " .. S.rollCount .. "  |  Rare Hits: " .. S.hitCount,
        "Best: " .. S.bestCard,
        string.rep("─", 44),
    }
    local lines = {}
    for _, h in ipairs(header)       do table.insert(lines, h) end
    for _, e in ipairs(S.rollHistory) do table.insert(lines, e) end
    local out = table.concat(lines, "\n")

    if setclipboard then
        setclipboard(out)
        log("📋 Copied " .. #S.rollHistory .. " entries to clipboard.", 130, 200, 255)
    else
        -- executor doesn't expose setclipboard — dump tail to visible log
        log("setclipboard unavailable. Last 10 entries:", 255, 160, 60)
        local start = math.max(1, #S.rollHistory - 9)
        for i = start, #S.rollHistory do
            log(S.rollHistory[i], 170, 170, 190)
        end
    end
end)
