-- ─── HELPERS ──────────────────────────────────────────────────────────────────
local function setStatus(txt, r, g, b)
    STxt.Text = txt
    SDot.BackgroundColor3 = Color3.fromRGB(r, g, b)
end

local function updateCount()
    RollCnt.Text = "Rolls: " .. S.rollCount .. "   |   Rare Hits: " .. S.hitCount
end

local function log(txt, r, g, b)
    r, g, b = r or 190, g or 190, b or 190
    local e = Instance.new("TextLabel")
    e.Size                   = UDim2.new(1, -6, 0, 0)
    e.AutomaticSize          = Enum.AutomaticSize.Y
    e.BackgroundTransparency = 1
    e.Text                   = txt
    e.TextColor3             = Color3.fromRGB(r, g, b)
    e.TextSize               = 11
    e.Font                   = Enum.Font.Code
    e.TextXAlignment         = Enum.TextXAlignment.Left
    e.TextWrapped            = true
    e.Parent                 = LogSF
    task.defer(function()
        LogSF.CanvasPosition = Vector2.new(0, LogSF.AbsoluteCanvasSize.Y)
    end)
    -- cap at 50 entries
    local kids = LogSF:GetChildren()
    if #kids > 52 then
        for i = 1, #kids - 50 do
            if kids[i]:IsA("TextLabel") then kids[i]:Destroy() end
        end
    end
end

local function checkRare(str)
    local low = str:lower()
    local bestW, bestS = nil, 0
    for word, score in pairs(CFG.RARE_WORDS) do
        if low:find(word) and score > bestS then
            bestW, bestS = word, score
        end
    end
    return bestW, bestS
end

local function processResult(...)
    local parts = {...}
    local combined = ""
    for _, v in ipairs(parts) do
        combined = combined .. tostring(v) .. " "
    end
    combined = combined:gsub("^%s+", ""):gsub("%s+$", "")
    if combined == "" then return end

    local word, score = checkRare(combined)
    if word then
        S.hitCount = S.hitCount + 1
        updateCount()
        local display = combined:sub(1, 55)
        log("★ " .. display, 255, 210, 0)
        if score > S.bestScore then
            S.bestScore  = score
            S.bestCard   = display
            BLbl.Text    = "🏆  Best: " .. display
        end
    else
        log("› " .. combined:sub(1, 55), 155, 155, 175)
    end
end

-- ─── DRAG (touch + mouse) ─────────────────────────────────────────────────────
do
    local dragging, dStart, wStart
    TBar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dStart   = i.Position
            wStart   = Win.Position
        end
    end)
    TBar.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseMove) then
            local d = i.Position - dStart
            Win.Position = UDim2.new(
                wStart.X.Scale, wStart.X.Offset + d.X,
                wStart.Y.Scale, wStart.Y.Offset + d.Y)
        end
    end)
    TBar.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
end

-- ─── TEST GAME ────────────────────────────────────────────────────────────────
BtnTest.MouseButton1Click:Connect(function()
    setStatus("Scanning game…", 255, 200, 0)
    log("── Scanning for remotes ──", 100, 140, 255)
    task.wait(0.05)

    local matched = {}
    local allFound = {}

    local function scan(parent)
        for _, obj in ipairs(parent:GetDescendants()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                table.insert(allFound, obj)
                local n = obj.Name:lower()
                for _, kw in ipairs(CFG.ROLL_KEYS) do
                    if n:find(kw) then
                        table.insert(matched, obj)
                        break
                    end
                end
            end
        end
    end
    scan(RS)
    scan(game.Workspace)

    log("Total remotes in game: " .. #allFound, 170, 170, 170)

    if #matched > 0 then
        -- best match picked automatically
        S.remote      = matched[1]
        S.remoteClass = matched[1].ClassName
        setStatus("✓ Compatible! Remote: " .. matched[1].Name, 55, 215, 75)
        log("✓ Roll remote: " .. matched[1].Name .. " [" .. matched[1].ClassName .. "]", 60, 220, 100)
        for i = 2, math.min(#matched, 5) do
            log("  alt: " .. matched[i].Name, 130, 200, 130)
        end
        -- hook result listener
        if matched[1]:IsA("RemoteEvent") then
            matched[1].OnClientEvent:Connect(function(...)
                processResult(...)
            end)
            log("  Result listener attached.", 120, 120, 160)
        end
    elseif #allFound > 0 then
        setStatus("⚠  Remotes exist, no match", 255, 155, 0)
        log("⚠ No roll keyword matched. All remotes:", 255, 160, 60)
        for i, r in ipairs(allFound) do
            if i > 15 then
                log("  … and " .. (#allFound - 15) .. " more", 140, 140, 140)
                break
            end
            log("  [" .. i .. "] " .. r.Name .. " (" .. r.ClassName .. ")", 200, 200, 200)
        end
        log("→ Auto Roll will try remote [1]: " .. allFound[1].Name, 200, 180, 100)
        S.remote      = allFound[1]
        S.remoteClass = allFound[1].ClassName
        if allFound[1]:IsA("RemoteEvent") then
            allFound[1].OnClientEvent:Connect(function(...)
                processResult(...)
            end)
        end
    else
        setStatus("✗  No remotes found", 215, 55, 55)
        log("✗ No RemoteEvents found in this game.", 215, 75, 75)
        log("  Game may use 100% local RNG — Hook RNG still works.", 155, 155, 155)
    end
end)

-- ─── HOOK RNG ─────────────────────────────────────────────────────────────────
BtnHook.MouseButton1Click:Connect(function()
    if S.hooked then
        log("math.random already hooked.", 160, 160, 160)
        return
    end
    local _orig = math.random
    math.random = function(a, b)
        if a and b then
            return b          -- integer range → always max (best outcome)
        elseif a then
            return a
        else
            return 0.99999    -- float call → near 1.0
        end
    end
    S.hooked = true
    BtnHook.BackgroundColor3 = Color3.fromRGB(35, 110, 35)
    BtnHook.Text = "✓  RNG Hooked"
    log("✓ math.random hooked → max forced", 90, 215, 90)
    log("  Only affects LOCAL RNG — server rolls unaffected.", 130, 130, 150)
end)

-- ─── AUTO ROLL ────────────────────────────────────────────────────────────────
BtnRoll.MouseButton1Click:Connect(function()
    S.autoRoll = not S.autoRoll

    if S.autoRoll then
        BtnRoll.BackgroundColor3 = Color3.fromRGB(155, 50, 35)
        BtnRoll.Text             = "⏸  Stop Roll"
        log("── Auto-roll ON ──", 90, 215, 90)

        if not S.remote then
            log("⚠ No remote detected yet — run Test Game first.", 255, 160, 50)
            log("  Will keep trying every roll tick.", 160, 160, 160)
        end

        task.spawn(function()
            while S.autoRoll do
                if S.remote then
                    pcall(function()
                        if S.remoteClass == "RemoteEvent" then
                            S.remote:FireServer()
                        else
                            S.remote:InvokeServer()
                        end
                    end)
                    S.rollCount = S.rollCount + 1
                    updateCount()
                else
                    -- remote not found yet, keep retrying scan silently
                    local r = RS:FindFirstChildWhichIsA("RemoteEvent", true)
                    if r then S.remote = r S.remoteClass = r.ClassName end
                end
                task.wait(CFG.ROLL_DELAY)
            end
        end)
    else
        BtnRoll.BackgroundColor3 = Color3.fromRGB(35, 115, 55)
        BtnRoll.Text             = "▶  Auto Roll"
        log("── Auto-roll OFF — total rolls: " .. S.rollCount .. " ──", 190, 190, 190)
    end
end)
